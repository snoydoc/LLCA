import pathlib

class FileManager:
    """A class for managing files in a specified directory."""

    def __init__(self, base: str) -> None:
        """Initialize the FileManager with a base directory.

        Args:
            base: The base directory path to manage files in.
        """
        self.base = pathlib.Path(base).expanduser().resolve()

    def read_file(self, name: str) -> str:
        """Read the contents of a file.

        Args:
            name: The name of the file to read.

        Returns:
            str: The content of the file.

        Raises:
            FileNotFoundError: If the file does not exist.
            PermissionError: If the user does not have permission to read the file.
            OSError: If other OS errors occur during reading.
        """
        file_path = self.base / name
        try:
            return file_path.read_text()
        except FileNotFoundError as e:
            raise FileNotFoundError(f"File '{name}' not found in {self.base}") from e
        except PermissionError as e:
            raise PermissionError(f"Permission denied to read '{name}'") from e
        except OSError as e:
            raise OSError(f"Error reading '{name}': {str(e)}") from e

    def write_file(self, name: str, data: str) -> None:
        """Write data to a file.

        Args:
            name: The name of the file to write.
            data: The content to write to the file.

        Raises:
            FileExistsError: If the file already exists and overwrite is not allowed.
            DirectoryNotFoundError: If the parent directory does not exist.
            PermissionError: If the user does not have permission to write the file.
            OSError: If other OS errors occur during writing.
        """
        file_path = self.base / name
        try:
            file_path.write_text(data)
        except FileExistsError:
            raise FileExistsError(f"File '{name}' already exists") from None
        except FileNotFoundError as e:
            raise FileNotFoundError(f"Parent directory not found: {self.base}") from e
        except PermissionError as e:
            raise PermissionError(f"Permission denied to write '{name}'") from e
        except OSError as e:
            raise OSError(f"Error writing to '{name}': {str(e)}") from e

    def list_files(self) -> list[str]:
        """List all files and directories in the base directory.

        Returns:
            list[str]: The names of files and directories in the base directory.

        Raises:
            FileNotFoundError: If the base directory does not exist.
            PermissionError: If the user does not have permission to list directory contents.
            OSError: If other OS errors occur during listing.
        """
        try:
            return [entry.name for entry in self.base.iterdir()]
        except FileNotFoundError as e:
            raise FileNotFoundError(f"Base directory '{self.base}' not found") from e
        except PermissionError as e:
            raise PermissionError(f"Permission denied to access '{self.base}'") from e
        except OSError as e:
            raise OSError(f"Error accessing '{self.base}': {str(e)}") from e
