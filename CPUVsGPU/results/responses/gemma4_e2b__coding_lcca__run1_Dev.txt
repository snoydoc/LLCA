from pathlib import Path
from typing import Union

class FileManager:
    """
    A class to manage file operations (reading, writing, listing) within a specified base directory.
    """
    def __init__(self, base: Union[str, Path]):
        """
        Initializes the FileManager with a base directory path.

        Args:
            base: The path to the base directory where files are managed.
        """
        self.base = Path(base).resolve()

    def read_file(self, filename: str) -> str:
        """
        Reads the content of a specified file.

        Args:
            filename: The name of the file to read from the base directory.

        Returns:
            The content of the file as a string.

        Raises:
            FileNotFoundError: If the specified file does not exist.
            IOError: If there is an issue reading the file.
        """
        file_path = self.base / filename
        if not file_path.is_file():
            raise FileNotFoundError(f"File not found: {file_path}")
        
        try:
            return file_path.read_text(encoding='utf-8')
        except IOError as e:
            raise IOError(f"Error reading file {file_path}: {e}")

    def write_file(self, filename: str, data: str):
        """
        Writes data to a specified file. Creates the file if it does not exist.

        Args:
            filename: The name of the file to write to.
            data: The string content to write to the file.

        Raises:
            IOError: If there is an issue writing to the file.
        """
        file_path = self.base / filename
        try:
            file_path.write_text(data, encoding='utf-8')
        except IOError as e:
            raise IOError(f"Error writing to file {file_path}: {e}")

    def list_files(self) -> list[str]:
        """
        Lists all files and directories within the base directory.

        Returns:
            A list of names (strings) of all entries in the base directory.

        Raises:
            FileNotFoundError: If the base directory does not exist.
            IOError: If there is an issue listing the directory contents.
        """
        if not self.base.is_dir():
            raise FileNotFoundError(f"Base directory does not exist: {self.base}")
            
        try:
            # Use iterdir() to list contents and extract just the names
            return [item.name for item in self.base.iterdir()]
        except IOError as e:
            raise IOError(f"Error listing files in {self.base}: {e}")
