import pathlib


class FileManager:
    def __init__(self, base: str | pathlib.Path) -> None:
        self.base = pathlib.Path(base)

    def read_file(self, name: str) -> str:
        """Read the contents of a file.

        Args:
            name: The name of the file to read.

        Returns:
            The contents of the file.

        Raises:
            FileNotFoundError: If the file does not exist.
        """
        file_path = self.base / name
        if not file_path.exists():
            raise FileNotFoundError(f"File {file_path} does not exist")
        return file_path.read_text()

    def write_file(self, name: str, data: str) -> None:
        """Write data to a file.

        Args:
            name: The name of the file to write.
            data: The data to write to the file.

        Raises:
            FileNotFoundError: If the parent directory does not exist.
            IOError: If there is an error writing the file.
        """
        file_path = self.base / name
        try:
            file_path.write_text(data)
        except FileNotFoundError:
            raise FileNotFoundError(f"Parent directory for {file_path} does not exist")
        except IOError as e:
            raise IOError(f"Error writing to file {file_path}: {str(e)}") from e

    def list_files(self) -> list[str]:
        """List all files in the base directory.

        Returns:
            A list of filenames in the base directory.

        Raises:
            NotADirectoryError: If the base path is not a directory.
        """
        if not self.base.is_dir():
            raise NotADirectoryError(f"{self.base} is not a directory")
        return [str(file_path.name) for file_path in self.base.iterdir()]
