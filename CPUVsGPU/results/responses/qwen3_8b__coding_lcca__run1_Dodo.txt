```python
from pathlib import Path

class FileManager:
    def __init__(self, base):
        """Initialize the FileManager with a base directory."""
        self.base_path = Path(base)

    def read_file(self, name):
        """Read the contents of a file.
        
        Args:
            name: The name of the file to read.
        
        Returns:
            The content of the file as a string.
        """
        file_path = self.base_path / name
        return file_path.read_text()

    def write_file(self, name, data):
        """Write data to a file.
        
        Args:
            name: The name of the file to write.
            data: The data to write to the file.
        """
        file_path = self.base_path / name
        file_path.write_text(data)

    def list_files(self):
        """List all files and directories in the base directory.
        
        Returns:
            A list of filenames.
        """
        return [f.name for f in self.base_path.iterdir()]
```
