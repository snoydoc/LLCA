```python
from pathlib import Path

class FileManager:
    def __init__(self, base):
        self.base = Path(base)

    def read_file(self, name):
        """
        Reads the content of a file.

        Args:
            name (str): The name of the file to read.

        Returns:
            str: The content of the file.

        Raises:
            FileNotFoundError: If the file does not exist.
        """
        file_path = self.base / name
        if not file_path.exists():
            raise FileNotFoundError(f"File '{name}' not found.")
        return file_path.read_text()

    def write_file(self, name, data):
        """
        Writes data to a file.

        Args:
            name (str): The name of the file to write.
            data (str): The data to write to the file.

        Raises:
            PermissionError: If the file cannot be written to.
        """
        file_path = self.base / name
        try:
            file_path.write_text(data)
        except PermissionError as e:
            raise PermissionError(f"Permission denied to write to '{name}': {e}")

    def list_files(self):
        """
        Lists all files in the directory.

        Returns:
            list: A list of file names.
        """
        return [file.name for file in self.base.iterdir() if file.is_file()]

    def list_directories(self):
        """
        Lists all directories in the directory.

        Returns:
            list: A list of directory names.
        """
        return [dir.name for dir in self.base.iterdir() if dir.is_dir()]
```
