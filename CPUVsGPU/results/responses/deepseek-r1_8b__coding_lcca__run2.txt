import pathlib


class FileManager:
    """A class to manage files within a specified base directory."""

    def __init__(self, base):
        """Initialize the FileManager with a base directory.

        Args:
            base (str): The path to the base directory.

        Raises:
            ValueError: If the base path does not exist or is not a directory.
        """
        self.base = pathlib.Path(base)
        if not self.base.exists():
            raise ValueError(f"The base directory '{base}' does not exist.")
        if not self.base.is_dir():
            raise ValueError(f"'{base}' is not a directory.")

    def read_file(self, name):
        """Read the contents of a file.

        Args:
            name (str): The name of the file to read.

        Returns:
            str: The contents of the file.

        Raises:
            FileNotFoundError: If the file does not exist.
            IOError: If there is an error reading the file.
        """
        file_path = self.base / name
        try:
            with open(file_path, 'r') as f:
                return f.read()
        except FileNotFoundError:
            raise FileNotFoundError(f"File '{name}' not found in '{self.base}'.")
        except IOError as e:
            raise IOError(f"Error reading file '{name}': {e}")

    def write_file(self, name, data):
        """Write data to a file, creating the directory if necessary.

        Args:
            name (str): The name of the file to write.
            data (str): The data to write to the file.

        Raises:
            IOError: If there is an error writing to the file or creating directories.
        """
        file_path = self.base / name
        try:
            parent_dir = file_path.parent
            if not parent_dir.exists():
                parent_dir.mkdir(parents=True, exist_ok=True)
            with open(file_path, 'w') as f:
                f.write(data)
        except IOError as e:
            raise IOError(f"Error writing file '{name}': {e}")

    def list_files(self):
        """List all files in the base directory.

        Returns:
            list: A list of filenames in the base directory.

        Raises:
            IOError: If there is an error accessing the directory.
        """
        try:
            return [f.name for f in self.base.iterdir()]
        except IOError as e:
            raise IOError(f"Error listing files in '{self.base}': {e}")
