from pathlib import Path

class FileManager:
    def __init__(self, base_path):
        """Initialize FileManager with a base path."""
        self.base_path = Path(base_path)
        
    def read_file(self, file_name):
        """Read and return the contents of a file.
        
        Args:
            file_name (str): Name of the file to read
            
        Returns:
            str: Contents of the file
            
        Raises:
            FileNotFoundError: If the file does not exist
        """
        file_path = self.base_path / file_name
        try:
            with open(file_path, 'r') as f:
                return f.read()
        except FileNotFoundError:
            raise FileNotFoundError(f"File '{file_name}' not found in directory '{self.base_path}'.")
    
    def write_file(self, file_name, data):
        """Write data to a file.
        
        Args:
            file_name (str): Name of the file to write to
            data (str): Data to write to the file
        """
        file_path = self.base_path / file_name
        # Ensure parent directories exist
        file_path.parent.mkdir(parents=True, exist_ok=True)
        try:
            with open(file_path, 'w') as f:
                f.write(data)
        except IOError:
            raise IOError(f"Error writing to file '{file_name}' in directory '{self.base_path}'.")
    
    def list_files(self):
        """List all files in the base directory.
        
        Returns:
            list: List of file names in the base directory
        """
        try:
            return [f.name for f in self.base_path.iterdir() if f.is_file()]
        except FileNotFoundError:
            raise FileNotFoundError(f"Base directory '{self.base_path}' does not exist.")
