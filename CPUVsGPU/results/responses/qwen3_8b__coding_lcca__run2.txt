```python
import pathlib

class FileManager:
    def __init__(self, base):
        """
        Initialize the FileManager with a base directory.
        
        Args:
            base (str): The base directory path.
        
        Raises:
            NotADirectoryError: If the provided base is not a valid directory.
        """
        self.base = pathlib.Path(base)
        if not self.base.is_dir():
            raise NotADirectoryError(f"{self.base} is not a valid directory")

    def read_file(self, name):
        """
        Read the contents of a file in the base directory.
        
        Args:
            name (str): The name of the file to read.
        
        Returns:
            str: The contents of the file.
        
        Raises:
            FileNotFoundError: If the file does not exist.
            PermissionError: If the file cannot be read.
        """
        file_path = self.base / name
        return file_path.open().read()

    def write_file(self, name, data):
        """
        Write data to a file in the base directory.
        
        Args:
            name (str): The name of the file to write.
            data (str): The data to write to the file.
        """
        file_path = self.base / name
        file_path.write_text(data)

    def list_files(self):
        """
        List all files and directories in the base directory.
        
        Returns:
            list: A list of strings representing the names of files and directories.
        """
        return [str(p) for p in self.base.iterdir()]
```
