```python
from datetime import date

def days_between(date1_str, date2_str):
  """
  Calculates the number of days between two dates given as strings in YYYY-MM-DD format.

  Args:
    date1_str: The first date string in YYYY-MM-DD format.
    date2_str: The second date string in YYYY-MM-DD format.

  Returns:
    The number of days between the two dates as an integer.
    Returns None if either input string is not in the correct format.
  """
  try:
    date1 = date.fromisoformat(date1_str)
    date2 = date.fromisoformat(date2_str)
    return abs((date2 - date1).days)
  except ValueError:
    return None

if __name__ == '__main__':
  # Usage example
  date_string1 = "2023-10-26"
  date_string2 = "2023-11-05"
  days = days_between(date_string1, date_string2)
  print(f"The number of days between {date_string1} and {date_string2} is: {days}")  # Output: 10

  date_string3 = "2024-01-01"
  date_string4 = "2023-12-31"
  days = days_between(date_string3, date_string4)
  print(f"The number of days between {date_string3} and {date_string4} is: {days}") # Output: 730

  # Example with invalid date format
  date_string5 = "2023/10/26"
  date_string6 = "2023-11-05"
  days = days_between(date_string5, date_string6)
  print(f"The number of days between {date_string5} and {date_string6} is: {days}") # Output: None
```
