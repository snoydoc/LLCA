from datetime import datetime

def days_between(date_str1: str, date_str2: str) -> int:
    """
    Calculates the absolute number of days between two given dates.

    The input dates must be in the format 'YYYY-MM-DD'.

    Args:
        date_str1: The first date string (YYYY-MM-DD).
        date_str2: The second date string (YYYY-MM-DD).

    Returns:
        The absolute number of days between the two dates.

    Raises:
        ValueError: If the date strings are not in the correct format.
    """
    # Define the format expected for parsing the input strings
    date_format = '%Y-%m-%d'

    try:
        # Convert the string dates into datetime objects
        date1 = datetime.strptime(date_str1, date_format)
        date2 = datetime.strptime(date_str2, date_format)

        # Calculate the difference (results in a timedelta object)
        time_difference = abs(date1 - date2)

        # Return the difference in days
        return time_difference.days

    except ValueError as e:
        # Handle cases where the input strings are not valid dates
        raise ValueError(f"Invalid date format. Please use YYYY-MM-DD. Error: {e}")
