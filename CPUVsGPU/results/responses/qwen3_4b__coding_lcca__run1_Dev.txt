```python
from pathlib import Path

class FileManager:
    def __init__(self, base: Path):
        self.base = base

    def read_file(self, name: str) -> str:
        """Read the content of a file from the base directory.

        Args:
            name: The name of the file to read.

        Returns:
            The content of the file as a string.

        Raises:
            FileNotFoundError: If the file does not exist.
        """
        file_path = self.base / name
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {file_path}")
        with open(file_path, 'r') as f:
            return f.read()

    def write_file(self, name: str, data: str) -> None:
        """Write data to a file in the base directory.

        Args:
            name: The name of the file to write.
            data: The string data to write.

        Raises:
            PermissionError: If there is a permission issue.
        """
        file_path = self.base / name
        file_path.parent.mkdir(parents=True, exist_ok=True)
        with open(file_path, 'w') as f:
            f.write(data)

    def list_files(self) -> list:
        """List all files and directories in the base directory.

        Returns:
            A list of strings representing the names of all entries in the base directory.
        """
        return [str(item) for item in self.base.iterdir()]
```
