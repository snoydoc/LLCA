import pathlib

class FileManager:
    def __init__(self, base):
        """Initialize FileManager with a base directory path."""
        self.base = pathlib.Path(base).resolve()
    
    def read_file(self, name):
        """Read and return the content of a file.
        
        Args:
            name (str): Name of the file to read.
            
        Returns:
            str: Content of the file.
            
        Raises:
            FileNotFoundError: If the file does not exist.
            PermissionError: If access is denied.
        """
        file_path = self.base / name
        return file_path.read_text()
    
    def write_file(self, name, data):
        """Write data to a file.
        
        Args:
            name (str): Name of the file to write to.
            data (str): Data to write to the file.
        """
        file_path = self.base / name
        file_path.write_text(data)
    
    def list_files(self):
        """List all files and directories in the base directory.
        
        Returns:
            list: List of names of all items in the base directory.
            
        Raises:
            FileNotFoundError: If the base directory does not exist.
        """
        if not self.base.exists():
            raise FileNotFoundError(f"Base directory {self.base} does not exist")
        return [item.name for item in self.base.iterdir()]
