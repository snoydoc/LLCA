```python
from pathlib import Path


class FileManager:
    """A class for managing files in a base directory."""

    def __init__(self, base: str) -> None:
        """
        Initialize the FileManager with a base directory path.

        Args:
            base (str): The base directory path

        Raises:
            NotADirectoryError: If the base path is not a valid directory
        """
        self.base_path = Path(base)
        if not self.base_path.is_dir():
            raise NotADirectoryError(f"{base} is not a directory")

    def read_file(self, name: str) -> str:
        """
        Read the contents of a file from the base directory.

        Args:
            name (str): The name of the file to read

        Returns:
            str: The contents of the file

        Raises:
            FileNotFoundError: If the file does not exist
            IsADirectoryError: If the path points to a directory
            PermissionError: If permission is denied
        """
        file_path = self.base_path / name
        if not file_path.exists():
            raise FileNotFoundError(f"File {name} not found in {self.base_path}")
        if file_path.is_dir():
            raise IsADirectoryError(f"{name} is a directory, not a file")
        return file_path.read_text()

    def write_file(self, name: str, data: str) -> None:
        """
        Write data to a file in the base directory.

        Args:
            name (str): The name of the file to write
            data (str): The data to write to the file

        Raises:
            FileExistsError: If the file already exists and write is not allowed
            PermissionError: If permission is denied
            OSError: For other I/O errors
        """
        file_path = self.base_path / name
        try:
            file_path.write_text(data)
        except FileExistsError:
            raise FileExistsError(f"File {name} already exists")
        except PermissionError as e:
            raise PermissionError(f"Permission denied for {name}") from e

    def list_files(self) -> list[str]:
        """
        List all files and directories in the base directory.

        Returns:
            list[str]: A list of filenames and directory names

        Note: This includes both files and directories
        """
        return [str(entry.name) for entry in self.base_path.iterdir()]
```
